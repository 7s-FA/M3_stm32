cmake_minimum_required(VERSION 3.16)
project(burger_stm32_bridge)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
set(deps rclcpp sensor_msgs geometry_msgs nav_msgs std_srvs message_filters tf2 tf2_ros)
foreach(dep IN LISTS deps)
 find_package(${dep} REQUIRED)
endforeach()
add_executable(stm32_adapter src/adapter.cpp)
ament_target_dependencies(stm32_adapter ${deps})
add_executable(stm32_odometry src/odometry_main.cpp src/odometry.cpp)
target_include_directories(stm32_odometry PRIVATE include)
ament_target_dependencies(stm32_odometry ${deps})
install(TARGETS stm32_adapter stm32_odometry DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_package()
